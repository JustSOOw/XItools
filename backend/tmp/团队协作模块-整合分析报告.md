 ├── team: Team

 ├── member: TeamMember
 ├── project: Project
 └── permission: 'view' | 'edit'

TaskComment (评论)
 ├── task: Task

 ├── content: string

router.put(
  '/teams/:teamId',
  authMiddleware,
  requireTeamOwner,
  async (request: FastifyRequest<{ Params: { teamId: string }; Body: UpdateTeamDTO }>, reply) => {
    const { teamId } = request.params;

#### 方案2：在前端添加事件监听

**文件**: `frontend/src/services/socketService.ts:174-267`

```typescript
private setupTeamEventListeners(): void {
  if (!this.socket) return;

  // ========================================
  // 团队信息变更事件
  // ========================================

  this.socket.on('team_updated', (data: { teamId: string; team: Team }) => {
    console.log('收到团队更新事件:', data);

    const currentTeam = useTeamStore.getState().currentTeam;

    // 如果是当前团队，更新store
    if (currentTeam && currentTeam.id === data.teamId) {
      useTeamStore.getState().updateTeamInfo(data.team);
    }
  });

  this.socket.on('team_dissolved', (data: { teamId: string }) => {
    console.log('收到团队解散事件:', data);

    // 清除团队信息，导航回个人工作区
    useTeamStore.getState().clearTeam();
    window.location.href = '/';
  });

  // ========================================
  // 成员角色变更事件
  // ========================================

  this.socket.on('team_member_role_changed', (data: { teamId: string; memberId: string; role: string }) => {
    console.log('收到成员角色变更事件:', data);

    const currentTeam = useTeamStore.getState().currentTeam;

    if (currentTeam && currentTeam.id === data.teamId) {
      useTeamStore.getState().fetchMembers(data.teamId);
    }
  });

  // ========================================
  // 邀请状态变更事件
  // ========================================

  this.socket.on('team_invitation_sent', (data: { teamId: string; invitations: TeamInvitation[] }) => {
    console.log('收到邀请发送事件:', data);

    const currentTeam = useTeamStore.getState().currentTeam;

    if (currentTeam && currentTeam.id === data.teamId) {
      useTeamStore.getState().fetchInvitations(data.teamId);
    }
  });

  this.socket.on('team_invitation_accepted', (data: { teamId: string; invitation: TeamInvitation }) => {
    console.log('收到邀请接受事件:', data);

    const currentTeam = useTeamStore.getState().currentTeam;

    if (currentTeam && currentTeam.id === data.teamId) {
      useTeamStore.getState().fetchInvitations(data.teamId);
      useTeamStore.getState().fetchMembers(data.teamId);
    }
  });

  // ========================================
  // 权限变更事件
  // ========================================

  this.socket.on('permission_granted', (data: { projectId: string; permission: ProjectPermission }) => {
    console.log('收到权限授予事件:', data);

    // 刷新权限列表（如果权限管理页面打开）
    window.dispatchEvent(new CustomEvent('permission_granted', { detail: data }));
  });

  this.socket.on('permission_revoked', (data: { projectId: string; permissionId: string }) => {
    console.log('收到权限撤销事件:', data);

    window.dispatchEvent(new CustomEvent('permission_revoked', { detail: data }));
  });
}
```
